show ip ospf neighbor

# 4. Authentication on RouterA/RouterB
show running-config interface Ethernet0/0

# 5. Route and LSDB inspection
show ip route ospf
show ip ospf database
show ip ospf border-routers

debug ip ospf packet
undebug all

! Cisco SW-LAN - Layer 2 switch (IOS / IOS XE style)
! Gi0/1 -> CORE-R1 Gi0/1 trunk
! Fa0/2 -> PC-USER (VLAN 10)
! Fa0/3 -> PC-SERVER (VLAN 20)
!
hostname SW-LAN
no ip domain-lookup
!
vlan 10
 name USERS
vlan 20
 name SERVERS
!
interface GigabitEthernet0/1
 description TRUNK-TO-CORE-R1-Gi0/1
 switchport mode trunk
 switchport trunk allowed vlan 10,20
 no shutdown
!
interface FastEthernet0/2
 description PC-USER-VLAN10
 switchport mode access
 switchport access vlan 10
 spanning-tree portfast
 no shutdown
!
interface FastEthernet0/3
 description PC-SERVER-VLAN20
 switchport mode access
 switchport access vlan 20
 spanning-tree portfast
 no shutdown
!
end
write memory

! show vlan brief
! show interfaces trunk
! show interfaces status

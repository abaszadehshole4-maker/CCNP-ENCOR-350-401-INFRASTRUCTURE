# R2 — intended VRRP Backup
# Verify syntax/support for your exact Cisco platform and software.
enable
configure terminal
hostname R2
interface GigabitEthernet0/0
 description Shared LAN connection
 ip address 192.168.10.2 255.255.255.0
 no shutdown
 vrrp 10 ip 192.168.10.254
 vrrp 10 priority 100
 vrrp 10 preempt
exit
end
write memory
